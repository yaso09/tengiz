# Docker Housekeeping Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a `tengiz cleanup` command that reclaims disk space by pruning stopped non-Tengiz containers, dangling images, unused volumes and unused networks — while never touching Tengiz-managed containers.

**Architecture:** A new `runtime.Prune(ctx, opts)` method on the existing `runtime.Manager` interface shells out to targeted `docker` commands with Go-side label filtering. Pure parsing helpers (`parsePSLines`, `isTengizManaged`, `parseIDs`, `parseBuildCacheReclaimable`, `parseReclaimedSpace`) are unit-tested so the logic works without a live Docker daemon. The CLI wraps `Prune` with a two-phase flow: a `--dry-run` preview, an interactive confirmation, then execution. `Store.PruneAllBuildLogs` reuses the existing build-log retention logic across all apps in the current env.

**Tech Stack:** Go 1.26, `os/exec` (docker CLI), Cobra, existing `config.Store` / `runtime.Manager` / `internal/cli/root.go` patterns. No new external dependencies.

## Global Constraints

```go
package runtime

import "strings"

type PruneOptions struct {
	DryRun bool
	All    bool
}

type PruneStats struct {
	Containers int
	Images     int
	Volumes    int
	Networks   int
	BuildCache int
}

func (s PruneStats) Total() int {
	return s.Containers + s.Images + s.Volumes + s.Networks + s.BuildCache
}

type PruneReport struct {
	Stats          PruneStats
	DryRun         bool
	ContainerIDs   []string
	ImageIDs       []string
	VolumeIDs      []string
	NetworkIDs     []string
	BuildCacheSize string
}

func (r PruneReport) Total() int {
	return r.Stats.Total()
}

type prunableContainer struct {
	ID     string
	Name   string
	Labels string
}

// parsePSLines parses `docker ps -a --format "{{.ID}}|{{.Names}}|{{.Labels}}"` output.
func parsePSLines(out string) []prunableContainer {
	var result []prunableContainer
	for _, line := range strings.Split(strings.TrimSpace(out), "\n") {
		if line == "" {
			continue
		}
		parts := strings.SplitN(line, "|", 3)
		c := prunableContainer{}
		if len(parts) > 0 {
			c.ID = strings.TrimSpace(parts[0])
		}
		if len(parts) > 1 {
			c.Name = strings.TrimSpace(parts[1])
		}
		if len(parts) > 2 {
			c.Labels = strings.TrimSpace(parts[2])
		}
		if c.ID != "" {
			result = append(result, c)
		}
	}
	return result
}

// isTengizManaged reports whether the docker labels string carries the tengiz-app label.
func isTengizManaged(labels string) bool {
	for _, part := range strings.Split(labels, ",") {
		kv := strings.SplitN(part, "=", 2)
		if len(kv) == 2 && strings.TrimSpace(kv[0]) == labelKey {
			return true
		}
	}
	return false
}

// parseIDs parses `--format "{{.ID}}"` or `--format "{{.Name}}"` output into non-empty lines.
func parseIDs(out string) []string {
	var ids []string
	for _, line := range strings.Split(strings.TrimSpace(out), "\n") {
		line = strings.TrimSpace(line)
		if line != "" {
			ids = append(ids, line)
		}
	}
	return ids
}

// parseReclaimedSpace extracts the "Total reclaimed space:" value from docker builder prune output.
func parseReclaimedSpace(out string) string {
	for _, line := range strings.Split(out, "\n") {
		line = strings.TrimSpace(line)
		if strings.HasPrefix(line, "Total reclaimed space:") {
			return strings.TrimSpace(strings.TrimPrefix(line, "Total reclaimed space:"))
		}
	}
	return ""
}

// parseBuildCacheReclaimable extracts the reclaimable value of the "Build Cache" row
// from `docker system df --format "{{.Type}}|{{.Reclaimable}}"` output.
func parseBuildCacheReclaimable(out string) string {
	for _, line := range strings.Split(strings.TrimSpace(out), "\n") {
		parts := strings.SplitN(line, "|", 2)
		if len(parts) == 2 && strings.TrimSpace(parts[0]) == "Build Cache" {
			return strings.TrimSpace(parts[1])
		}
	}
	return ""
}
```

```go
package runtime

import (
	"context"
	"fmt"
	"log"
	"os/exec"
	"strings"
)

func (r *dockerRuntime) listNonTengizStoppedContainers(ctx context.Context) ([]string, error) {
	cmd := exec.CommandContext(ctx, "docker", "ps", "-a",
		"--filter", "status=exited",
		"--filter", "status=dead",
		"--format", "{{.ID}}|{{.Names}}|{{.Labels}}")
	out, err := cmd.CombinedOutput()
	if err != nil {
		return nil, fmt.Errorf("docker ps: %w", err)
	}
	var ids []string
	for _, c := range parsePSLines(string(out)) {
		if !isTengizManaged(c.Labels) {
			ids = append(ids, c.ID)
		}
	}
	return ids, nil
}

func (r *dockerRuntime) listDanglingImages(ctx context.Context) ([]string, error) {
	cmd := exec.CommandContext(ctx, "docker", "images",
		"--filter", "dangling=true",
		"--format", "{{.ID}}")
	out, err := cmd.CombinedOutput()
	if err != nil {
		return nil, fmt.Errorf("docker images: %w", err)
	}
	return parseIDs(string(out)), nil
}

func (r *dockerRuntime) listDanglingVolumes(ctx context.Context) ([]string, error) {
	cmd := exec.CommandContext(ctx, "docker", "volume", "ls",
		"--filter", "dangling=true",
		"--format", "{{.Name}}")
	out, err := cmd.CombinedOutput()
	if err != nil {
		return nil, fmt.Errorf("docker volume ls: %w", err)
	}
	return parseIDs(string(out)), nil
}

func (r *dockerRuntime) listUnusedNetworks(ctx context.Context) ([]string, error) {
	cmd := exec.CommandContext(ctx, "docker", "network", "ls",
		"--filter", "dangling=true",
		"--format", "{{.Name}}")
	out, err := cmd.CombinedOutput()
	if err != nil {
		return nil, fmt.Errorf("docker network ls: %w", err)
	}
	return parseIDs(string(out)), nil
}

func (r *dockerRuntime) buildCacheSize(ctx context.Context) (string, error) {
	cmd := exec.CommandContext(ctx, "docker", "system", "df",
		"--format", "{{.Type}}|{{.Reclaimable}}")
	out, err := cmd.CombinedOutput()
	if err != nil {
		return "", fmt.Errorf("docker system df: %w", err)
	}
	return parseBuildCacheReclaimable(string(out)), nil
}

func (r *dockerRuntime) pruneBuildCache(ctx context.Context) (string, error) {
	cmd := exec.CommandContext(ctx, "docker", "builder", "prune", "-af")
	out, err := cmd.CombinedOutput()
	if err != nil {
		return "", fmt.Errorf("docker builder prune: %w", err)
	}
	return parseReclaimedSpace(string(out)), nil
}

func (r *dockerRuntime) Prune(ctx context.Context, opts PruneOptions) (PruneReport, error) {
	report := PruneReport{DryRun: opts.DryRun}

	containers, err := r.listNonTengizStoppedContainers(ctx)
	if err != nil {
		return report, err
	}
	report.Stats.Containers = len(containers)
	report.ContainerIDs = containers
	if !opts.DryRun {
		for _, id := range containers {
			if err := r.Remove(ctx, id); err != nil {
				log.Printf("[runtime] prune container %s: %v", id, err)
			}
		}
	}

	images, err := r.listDanglingImages(ctx)
	if err != nil {
		return report, err
	}
	report.Stats.Images = len(images)
	report.ImageIDs = images
	if !opts.DryRun {
		for _, id := range images {
			if err := r.RemoveImage(ctx, id); err != nil {
				log.Printf("[runtime] prune image %s: %v", id, err)
			}
		}
	}

	volumes, err := r.listDanglingVolumes(ctx)
	if err != nil {
		return report, err
	}
	report.Stats.Volumes = len(volumes)
	report.VolumeIDs = volumes
	if !opts.DryRun {
		for _, name := range volumes {
			cmd := exec.CommandContext(ctx, "docker", "volume", "rm", name)
			if out, err := cmd.CombinedOutput(); err != nil {
				log.Printf("[runtime] prune volume %s: %v\n%s", name, err, string(out))
			}
		}
	}

	networks, err := r.listUnusedNetworks(ctx)
	if err != nil {
		return report, err
	}
	report.Stats.Networks = len(networks)
	report.NetworkIDs = networks
	if !opts.DryRun {
		for _, name := range networks {
			cmd := exec.CommandContext(ctx, "docker", "network", "rm", name)
			if out, err := cmd.CombinedOutput(); err != nil {
				log.Printf("[runtime] prune network %s: %v\n%s", name, err, string(out))
			}
		}
	}

	if opts.All {
		if opts.DryRun {
			size, err := r.buildCacheSize(ctx)
			if err != nil {
				return report, err
			}
			report.BuildCacheSize = size
			if size != "" && size != "0B" {
				report.Stats.BuildCache = 1
			}
		} else {
			reclaimed, err := r.pruneBuildCache(ctx)
			if err != nil {
				return report, err
			}
			report.BuildCacheSize = reclaimed
			if reclaimed != "" && reclaimed != "0B" {
				report.Stats.BuildCache = 1
			}
		}
	}

	return report, nil
}
```

```go
func (s *Store) PruneAllBuildLogs(keep int) (int, error) {
	s.mu.Lock()
	defer s.mu.Unlock()

	base := filepath.Join(s.dataDir, "build-logs", s.env)
	entries, err := os.ReadDir(base)
	if err != nil {
		if os.IsNotExist(err) {
			return 0, nil
		}
		return 0, fmt.Errorf("list build-log apps: %w", err)
	}

	total := 0
	for _, e := range entries {
		if !e.IsDir() {
			continue
		}
		n, err := pruneBuildLogDir(filepath.Join(base, e.Name()), keep)
		if err != nil {
			return total, err
		}
		total += n
	}
	return total, nil
}

// pruneBuildLogDir removes all but the newest `keep` .log files in dir,
// oldest-first. Returns the number of files removed.
func pruneBuildLogDir(dir string, keep int) (int, error) {
	entries, err := os.ReadDir(dir)
	if err != nil {
		if os.IsNotExist(err) {
			return 0, nil
		}
		return 0, fmt.Errorf("prune list: %w", err)
	}

	type logFile struct {
		name string
		info os.FileInfo
	}
	var files []logFile
	for _, e := range entries {
		if e.IsDir() || !strings.HasSuffix(e.Name(), ".log") {
			continue
		}
		info, err := e.Info()
		if err != nil {
			continue
		}
		files = append(files, logFile{name: e.Name(), info: info})
	}

	sort.Slice(files, func(i, j int) bool {
		return files[i].info.ModTime().Before(files[j].info.ModTime())
	})

	removed := 0
	for _, f := range files[:len(files)-keep] {
		if err := os.Remove(filepath.Join(dir, f.name)); err != nil {
			return removed, fmt.Errorf("remove build log: %w", err)
		}
		removed++
	}
	return removed, nil
}
```

Note: `files[:len(files)-keep]` is safe — when `len(files) <= keep` the slice is empty and the loop body never runs (the original code's early `return nil` is no longer needed). Behavior of `PruneBuildLogs` is unchanged: it sorts by `ModTime` ascending and removes the oldest files first.

```go
var cleanupCmd = &cobra.Command{
	Use:   "cleanup",
	Short: "Remove unused Docker resources",
	Long:  "Removes stopped containers that are not managed by Tengiz, dangling images, unused volumes and unused networks. Tengiz-managed containers are always protected. Use --dry-run to preview and --all for aggressive cleanup including the Docker build cache.",
	Args:  cobra.NoArgs,
	RunE: func(cmd *cobra.Command, args []string) error {
		env := getEnv(cmd)
		dryRun, _ := cmd.Flags().GetBool("dry-run")
		all, _ := cmd.Flags().GetBool("all")
		force, _ := cmd.Flags().GetBool("force")
		keepLogs, _ := cmd.Flags().GetInt("keep-build-logs")

		rt, err := runtime.NewDocker()
		if err != nil {
			return fmt.Errorf("docker: %w", err)
		}

		preview, err := rt.Prune(context.Background(), runtime.PruneOptions{DryRun: true, All: all})
		if err != nil {
			return fmt.Errorf("cleanup: %w", err)
		}
		fmt.Print(pruneSummary(preview))

		if preview.Total() == 0 || dryRun {
			return nil
		}

		if !force {
			fmt.Print("Continue? [y/N]: ")
			reader := bufio.NewReader(os.Stdin)
			answer, _ := reader.ReadString('\n')
			answer = strings.ToLower(strings.TrimSpace(answer))
			if answer != "y" && answer != "yes" {
				fmt.Println("Cleanup cancelled.")
				return nil
			}
		}

		execReport, err := rt.Prune(context.Background(), runtime.PruneOptions{All: all})
		if err != nil {
			return fmt.Errorf("cleanup: %w", err)
		}
		fmt.Print(pruneSummary(execReport))

		store := config.NewStoreWithEnv(dataDir, env)
		removedLogs, logErr := store.PruneAllBuildLogs(keepLogs)
		if logErr != nil {
			log.Printf("[tengiz] warning: prune build logs: %v", logErr)
		}
		if removedLogs > 0 {
			fmt.Printf("Removed %d old build log(s).\n", removedLogs)
		}
		return nil
	},
}

func pruneSummary(report runtime.PruneReport) string {
	verb := "would remove"
	if !report.DryRun {
		verb = "removed"
	}
	s := report.Stats
	var b strings.Builder
	fmt.Fprintf(&b, "Cleanup %s:\n", verb)
	fmt.Fprintf(&b, "  containers: %d\n", s.Containers)
	fmt.Fprintf(&b, "  images:     %d\n", s.Images)
	fmt.Fprintf(&b, "  volumes:    %d\n", s.Volumes)
	fmt.Fprintf(&b, "  networks:   %d\n", s.Networks)
	if report.BuildCacheSize != "" {
		fmt.Fprintf(&b, "  build cache: %s\n", report.BuildCacheSize)
	}
	return b.String()
}
```
