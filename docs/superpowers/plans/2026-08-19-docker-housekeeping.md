# Docker Housekeeping Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a `tengiz cleanup` command that reclaims disk space by pruning orphaned Tengiz-managed containers, unused `tengiz-apps/` images, and the Docker build cache — using label-based filters so user's unrelated Docker objects are never touched.

**Architecture:** The `runtime` package gains list + prune primitives built on the existing `os/exec` docker CLI pattern (`docker ps -a --filter label=tengiz-app`, `docker images --filter reference=tengiz-apps/*`, `docker builder prune`). Pure helper functions carry the filter logic so they are unit-testable without Docker (matching the repo's existing `buildRunArgs` pattern). The CLI composes these primitives into `tengiz cleanup`: it builds a "protected" set from the env-scoped store (apps, deployment history, previews), computes what is removable, and defaults to dry-run so nothing is deleted unless `--yes` is passed.

**Tech Stack:** Go 1.26, Cobra (CLI), existing `internal/runtime` (`Manager` interface, `dockerRuntime`, `stubManager`), existing `internal/config.Store`, existing `internal/types`. No new external dependencies.

## Global Constraints

- Only touch Docker resources labeled `tengiz-app` or built under the `tengiz-apps/` repository — never the user's other containers/images/volumes
- Never remove a **running** container (any container whose status starts with `Up`)
- Never remove an image whose tag ends in `-latest`
- Image pruning is env-scoped: only images whose tag matches the current env are candidates (tag prefix `<env>-` via `MatchesEnv`) — prevents a `--env staging` cleanup from deleting production rollback images
- Orphaned preview images (tag prefix `pr-`) are excluded from pruning candidates — they are cleaned up by the preview lifecycle instead (safety over aggression)
- `tengiz cleanup` defaults to dry-run; `--yes` is required to actually delete
- Select categories with `--containers`, `--images`, `--build-cache`; when none are given, all three run
- Container pruning does NOT filter by env label — the running-container rule plus the store protection set is sufficient safety for orphaned debris
- No new external dependencies; reuse existing `runtime.Remove` and `runtime.RemoveImage`
- Existing `KeepLastNImages` already caps per-app image count to 5; cleanup only removes images not referenced by `apps.json`, `deployments.json`, or `previews.json`
- Follow repo test conventions: pure helpers get unit tests, exec wrappers are thin and verified via `go build` + `go vet` + full test suite
- Per AGENTS.md: create a `feat/<name>` branch, add/update tests on every change, pass tests, commit after each task, update README on CLI/UX changes
- `docs/FUTURES_FEATURES.md` is auto-generated by the daily analysis workflow — do not edit it manually

```go
type ImageInfo struct {
	RepoTag string
	Tag     string
	ID      string
	Size    string
}

type ContainerInfo struct {
	ID     string
	Name   string
	Status string
}

func parseImageLine(line string) (ImageInfo, bool) {
	parts := strings.Split(line, "|")
	if len(parts) != 3 {
		return ImageInfo{}, false
	}
	repoTag, id, size := parts[0], parts[1], parts[2]
	idx := strings.LastIndex(repoTag, ":")
	if repoTag == "" || idx < 0 {
		return ImageInfo{}, false
	}
	return ImageInfo{
		RepoTag: repoTag,
		Tag:     repoTag[idx+1:],
		ID:      id,
		Size:    size,
	}, true
}

func parseContainerLine(line string) (ContainerInfo, bool) {
	parts := strings.Split(line, "|")
	if len(parts) != 3 {
		return ContainerInfo{}, false
	}
	name := strings.TrimPrefix(parts[1], "/")
	if name == "" {
		return ContainerInfo{}, false
	}
	return ContainerInfo{ID: parts[0], Name: name, Status: parts[2]}, true
}

func MatchesEnv(tag, env string) bool {
	if env == "" || env == "production" {
		env = "production"
	}
	return strings.HasPrefix(tag, env+"-") || strings.HasSuffix(tag, "-latest")
}

func RemovableContainers(containers []ContainerInfo, protect map[string]bool) []ContainerInfo {
	var result []ContainerInfo
	for _, c := range containers {
		if strings.HasPrefix(c.Status, "Up") {
			continue
		}
		if protect[c.Name] {
			continue
		}
		result = append(result, c)
	}
	return result
}

func RemovableImages(images []ImageInfo, protectTags map[string]bool) []ImageInfo {
	var result []ImageInfo
	for _, img := range images {
		if strings.HasSuffix(img.Tag, "-latest") {
			continue
		}
		if protectTags[img.RepoTag] {
			continue
		}
		result = append(result, img)
	}
	return result
}
```

```go
func (r *dockerRuntime) ListContainers(ctx context.Context) ([]ContainerInfo, error) {
	cmd := exec.CommandContext(ctx, "docker", "ps", "-a",
		"--filter", fmt.Sprintf("label=%s", labelKey),
		"--format", "{{.ID}}|{{.Names}}|{{.Status}}")
	out, err := cmd.CombinedOutput()
	if err != nil {
		return nil, fmt.Errorf("docker ps: %w\n%s", err, string(out))
	}
	var result []ContainerInfo
	for _, line := range strings.Split(strings.TrimSpace(string(out)), "\n") {
		if c, ok := parseContainerLine(line); ok {
			result = append(result, c)
		}
	}
	return result, nil
}

func (r *dockerRuntime) ListImages(ctx context.Context) ([]ImageInfo, error) {
	cmd := exec.CommandContext(ctx, "docker", "images",
		"--filter", "reference=tengiz-apps/*",
		"--format", "{{.Repository}}:{{.Tag}}|{{.ID}}|{{.Size}}")
	out, err := cmd.CombinedOutput()
	if err != nil {
		return nil, fmt.Errorf("docker images: %w\n%s", err, string(out))
	}
	var result []ImageInfo
	for _, line := range strings.Split(strings.TrimSpace(string(out)), "\n") {
		if img, ok := parseImageLine(line); ok {
			result = append(result, img)
		}
	}
	return result, nil
}
```

```go
func (r *dockerRuntime) PruneContainers(ctx context.Context, protect map[string]bool) (int, error) {
	containers, err := r.ListContainers(ctx)
	if err != nil {
		return 0, err
	}
	removed := 0
	for _, c := range RemovableContainers(containers, protect) {
		if err := r.Remove(ctx, c.Name); err != nil {
			log.Printf("[runtime] failed to remove container %s: %v", c.Name, err)
			continue
		}
		removed++
	}
	return removed, nil
}

func (r *dockerRuntime) PruneImages(ctx context.Context, protectTags map[string]bool, env string) (int, error) {
	images, err := r.ListImages(ctx)
	if err != nil {
		return 0, err
	}
	var candidates []ImageInfo
	for _, img := range images {
		if MatchesEnv(img.Tag, env) {
			candidates = append(candidates, img)
		}
	}
	removed := 0
	for _, img := range RemovableImages(candidates, protectTags) {
		if err := r.RemoveImage(ctx, img.RepoTag); err != nil {
			log.Printf("[runtime] failed to remove image %s: %v", img.RepoTag, err)
			continue
		}
		removed++
	}
	return removed, nil
}

func (r *dockerRuntime) PruneBuildCache(ctx context.Context, dryRun bool) (string, error) {
	args := []string{"builder", "prune"}
	if dryRun {
		args = append(args, "--dry-run")
	} else {
		args = append(args, "-f", "--format", "{{.SpaceReclaimed}}")
	}
	cmd := exec.CommandContext(ctx, "docker", args...)
	out, err := cmd.CombinedOutput()
	if err != nil {
		return "", fmt.Errorf("docker builder prune: %w\n%s", err, string(out))
	}
	return strings.TrimSpace(string(out)), nil
}
```

```go
package cli

import (
	"github.com/yaso09/tengiz/internal/config"
	"github.com/yaso09/tengiz/internal/runtime"
)

func buildProtectionSet(store *config.Store, env string) (map[string]bool, map[string]bool) {
	protectContainers := make(map[string]bool)
	protectTags := make(map[string]bool)

	apps, err := store.ListApps()
	if err == nil {
		for _, app := range apps {
			base := runtime.ContainerName(app.Name, env)
			protectContainers[base] = true
			if app.DeploymentSuffix != "" {
				protectContainers[base+"-"+app.DeploymentSuffix] = true
			}
			if app.ImageTag != "" {
				protectTags[app.ImageTag] = true
			}
			deps, depErr := store.GetDeployments(app.Name)
			if depErr == nil {
				for _, d := range deps {
					if d.ImageTag != "" {
						protectTags[d.ImageTag] = true
					}
				}
			}
		}
	}

	previews, err := store.ListAllPreviews()
	if err == nil {
		for _, pv := range previews {
			if pv.ContainerName != "" {
				protectContainers[pv.ContainerName] = true
			}
			if pv.ImageTag != "" {
				protectTags[pv.ImageTag] = true
			}
		}
	}

	return protectContainers, protectTags
}
```

```go
var cleanupCmd = &cobra.Command{
	Use:   "cleanup",
	Short: "Remove unused Docker resources to reclaim disk space",
	Long: `Clean up Docker resources managed by Tengiz:
  - orphaned containers (tengiz-app labeled, no longer tracked by the store)
  - unused tengiz-apps images (not referenced by any app, deployment, or preview)
  - Docker build cache

Only Tengiz-managed resources are touched. Other containers and images are left alone.

By default this runs in dry-run mode and only prints what would be removed.
Pass --yes to actually perform the cleanup.`,
	RunE: func(cmd *cobra.Command, args []string) error {
		env := getEnv(cmd)
		dryRun, _ := cmd.Flags().GetBool("dry-run")
		yes, _ := cmd.Flags().GetBool("yes")
		doContainers, _ := cmd.Flags().GetBool("containers")
		doImages, _ := cmd.Flags().GetBool("images")
		doBuildCache, _ := cmd.Flags().GetBool("build-cache")
		if !doContainers && !doImages && !doBuildCache {
			doContainers, doImages, doBuildCache = true, true, true
		}
		effectiveDryRun := dryRun || !yes

		rt, err := runtime.NewDocker()
		if err != nil {
			return fmt.Errorf("docker: %w", err)
		}
		store := config.NewStoreWithEnv(dataDir, env)
		protectContainers, protectTags := buildProtectionSet(store, env)
		ctx := cmd.Context()

		if effectiveDryRun {
			fmt.Println("[tengiz] dry-run: nothing will be removed. Use --yes to execute.")
		}

		if doContainers {
			containers, err := rt.ListContainers(ctx)
			if err != nil {
				return fmt.Errorf("list containers: %w", err)
			}
			removable := runtime.RemovableContainers(containers, protectContainers)
			if len(removable) == 0 {
				fmt.Println("containers: nothing to remove")
			}
			for _, c := range removable {
				if effectiveDryRun {
					fmt.Printf("would remove container %s (%s)\n", c.Name, c.Status)
				}
			}
			if !effectiveDryRun {
				n, err := rt.PruneContainers(ctx, protectContainers)
				if err != nil {
					return fmt.Errorf("prune containers: %w", err)
				}
				fmt.Printf("containers: removed %d\n", n)
			}
		}

		if doImages {
			images, err := rt.ListImages(ctx)
			if err != nil {
				return fmt.Errorf("list images: %w", err)
			}
			var candidates []runtime.ImageInfo
			for _, img := range images {
				if runtime.MatchesEnv(img.Tag, env) {
					candidates = append(candidates, img)
				}
			}
			removable := runtime.RemovableImages(candidates, protectTags)
			if len(removable) == 0 {
				fmt.Println("images: nothing to remove")
			}
			for _, img := range removable {
				if effectiveDryRun {
					fmt.Printf("would remove image %s (%s)\n", img.RepoTag, img.Size)
				}
			}
			if !effectiveDryRun {
				n, err := rt.PruneImages(ctx, protectTags, env)
				if err != nil {
					return fmt.Errorf("prune images: %w", err)
				}
				fmt.Printf("images: removed %d\n", n)
			}
		}

		if doBuildCache {
			out, err := rt.PruneBuildCache(ctx, effectiveDryRun)
			if err != nil {
				return fmt.Errorf("prune build cache: %w", err)
			}
			if out != "" {
				fmt.Println(out)
			}
		}

		return nil
	},
}
```
